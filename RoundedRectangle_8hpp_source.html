338 std::vector<glm::vec2> CreateCircleRegion(const glm::vec2 &position, float radius, float startAngle, float fillAngle) const
339 {
340 std::vector<glm::vec2> border;
341
342 float deltaAngle = glm::radians(fillAngle) / (float)m_PointsPerCorner;
343 float currentAngle = glm::radians(startAngle) + glm::radians(180.0f);
344
345 const glm::vec2 centre = {position.x, position.y};
346
347 for (uint32_t i = 0; i < m_PointsPerCorner; i++)
348 {
349 glm::vec2 point = {glm::cos(currentAngle) * radius + position.x, glm::sin(currentAngle) * radius + position.y};
350
351 border.push_back(point);
352 currentAngle += deltaAngle;
353 }
354
355 return border;
356 }
