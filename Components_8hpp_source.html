27 inline glm::mat4 CreateTransformation()
28 {
29 glm::mat4 transformation(1.0f);
30
31 // apply translation
32 transformation = glm::translate(transformation, Position);
33
34 // apply rotation
35 transformation = glm::rotate(transformation, glm::radians(Rotation.x), glm::vec3(1.0f, 0.0f, 0.0f));
36 transformation = glm::rotate(transformation, glm::radians(Rotation.y), glm::vec3(0.0f, 1.0f, 0.0f));
37 transformation = glm::rotate(transformation, glm::radians(Rotation.z), glm::vec3(0.0f, 0.0f, 1.0f));
38
39 // apply scale
40 transformation = glm::scale(transformation, Scale);
41
42 return transformation;
43 }
