163 inline ShaderResource ReflectedShaderResourceToShaderResource(const ReflectedResource &resource, ShaderStage stage)
164 {
165 ShaderResource output;
166
167 switch (resource.Type)
168 {
169 case ReflectedShaderDataType::StorageImage:
170 {
171 output.Type = ResourceType::StorageImage;
172 break;
173 }
174 case ReflectedShaderDataType::Texture:
175 {
176 output.Type = ResourceType::Texture;
177 break;
178 }
179 case ReflectedShaderDataType::UniformTextureBuffer:
180 {
181 output.Type = ResourceType::UniformTextureBuffer;
182 break;
183 }
184 case ReflectedShaderDataType::StorageTextureBuffer:
185 {
186 output.Type = ResourceType::StorageTextureBuffer;
187 break;
188 }
189 case ReflectedShaderDataType::Sampler:
190 {
191 output.Type = ResourceType::Sampler;
192 break;
193 }
194 case ReflectedShaderDataType::ComparisonSampler:
195 {
196 output.Type = ResourceType::ComparisonSampler;
197 break;
198 }
199 case ReflectedShaderDataType::CombinedImageSampler:
200 {
201 output.Type = ResourceType::CombinedImageSampler;
202 break;
203 }
204 case ReflectedShaderDataType::UniformBuffer:
205 {
206 output.Type = ResourceType::UniformBuffer;
207 break;
208 }
209 case ReflectedShaderDataType::StorageBuffer:
210 {
211 output.Type = ResourceType::StorageBuffer;
212 break;
213 }
214 case ReflectedShaderDataType::AccelerationStructure:
215 {
216 output.Type = ResourceType::AccelerationStructure;
217 break;
218 }
219 case ReflectedShaderDataType::PushConstants:
220 {
221 output.Type = ResourceType::PushConstants;
222 break;
223 }
224 default: throw std::runtime_error("Failed to find a valid resource type");
225 }
226
227 output.Access = resource.ResourceAccess;
228 output.Name = !resource.BlockName.empty() ? resource.BlockName : resource.InstanceName;
229 output.Set = resource.DescriptorSet;
230 output.Binding = resource.BindingPoint;
231 output.ResourceCount = resource.BindingCount;
232 output.RegisterSpace = resource.RegisterSpace;
233 output.Stage.AddFlag(stage);
234 output.Dimension = resource.Dimension;
235
236 return output;
237 }

249
250 std::map<std::string, ShaderResource> GetRequiredShaderResources() const
251 {
252 std::map<std::string, ShaderResource> requiredResources;
253
254 std::vector<Ref<IShaderModule>> shaderStages = GetShaderStages();
255 for (Ref<IShaderModule> module : shaderStages)
256 {
257 ShaderReflectionData reflectionData = module->Reflect();
258 for (const auto &resource : reflectionData.Resources)
259 {
260 std::string resourceName = !resource.BlockName.empty() ? resource.BlockName : resource.InstanceName;
261
262 if (requiredResources.find(resourceName) != requiredResources.end())
263 {
264 ShaderResource &requiredResource = requiredResources.at(resourceName);
265 ShaderResource newResource = ReflectedShaderResourceToShaderResource(resource, module->GetShaderStage());
266
267 if (newResource == requiredResource)
268 {
269 throw std::runtime_error("Shader resources that share the same name must match across stages");
270 }
271
272 requiredResource.Stage.AddFlag(module->GetShaderStage());
273 }
274 else
275 {
276 ShaderResource newResource = ReflectedShaderResourceToShaderResource(resource, module->GetShaderStage());
277 requiredResources[resourceName] = newResource;
278 }
279 }
280 }
281
282 return requiredResources;
283 }
